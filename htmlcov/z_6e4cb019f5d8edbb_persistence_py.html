26 """Nudges toward memory persistence on tool failure and at session end."""

28 def on_hook(self, hook_name: str, *, logger: logging.Logger, **kwargs: object) -> HookResult | None: 

29 """Emit a persist-to-memory nudge on tool failure or session end. 

30 

31 Args: 

32 hook_name: The hook event or plugin-scope name. 

33 logger: This plugin's hook-scoped child logger. 

34 **kwargs: Hook-specific keyword arguments. 

35 

36 Returns: 

37 A HookResult carrying the relevant nudge, or None. 

38 """ 

39 if hook_name == PluginScope.TOOL_FAILED: 

40 logger.debug("Fired persist-to-memory nudge after tool failure") 

41 return HookResult(notes=[_FAILURE_PERSIST_NOTE]) 

42 if hook_name == CanonicalHook.POST_TOOL_USE: 

43 tool_name = kwargs.get("tool_name") 

44 parameters = kwargs.get("parameters") 

45 task_id = kwargs.get("task_id") 

46 if (

51 and not has_memory_writes(task_id) 

52 ): 

53 logger.debug("Fired no-memory-writes warning at session end") 

54 return HookResult(notes=[_MEMORY_WARNING])
