60def should_remind(turn_count: int) -> bool: 

61 """Check whether the current turn count should trigger a scope reminder. 

62 

63 Triggers at the threshold, then every REMINDER_INTERVAL turns after. 

64 

65 Args: 

66 turn_count: The current turn count. 

67 

68 Returns: 

69 True if a scope-check reminder should be shown. 

70 """

73 return (turn_count - _SCOPE_CHECK_THRESHOLD) % _REMINDER_INTERVAL == 0

208def _parse_diff_stat_line(line: str) -> int: 

209 """Extract inserted+deleted line count from a single git diff --stat output line. 

210 

211 Args: 

212 line: A single line from git diff --stat output. 

213 

214 Returns: 

215 Total line count for this stat line. 

216 """ 

217 total = 0 

218 for part in line.split(","): 

219 stripped = part.strip() 

220 if "insertion" in stripped or "deletion" in stripped: 

221 with contextlib.suppress(ValueError, IndexError): 

222 total += int(stripped.split()[0]) 

223 return total

226def _get_diff_line_count(workspace_roots: list[str]) -> int: 

227 """Return total added+removed lines in the working tree of the first valid repo. 

228 

229 Args: 

230 workspace_roots: Workspace root paths to search for a git repo. 

231 

232 Returns: 

233 Total diff line count, or 0 if no repo or no diff. 

234 """ 

235 for root in workspace_roots: 

236 try: 

237 repo = git.Repo(root) 

238 diff = repo.git.diff("--stat", "HEAD") 

239 except ( 

240 git.exc.InvalidGitRepositoryError, 

241 git.exc.GitCommandError, 

242 git.exc.NoSuchPathError, 

243 ): 

244 continue 

245 else: 

246 return sum(_parse_diff_stat_line(line) for line in diff.splitlines()) 

247 return 0
