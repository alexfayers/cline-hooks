25def _run(payload: dict[str, object], env: dict[str, str] | None = None) -> list[str]: 

26 """Feed one payload through the entry point and capture what it printed. 

27 

28 Returns: 

29 Everything written to stdout, in order. 

30 """ 

31 output: list[str] = [] 

32 with ( 

33 patch("builtins.input", return_value=json.dumps(payload)), 

34 patch("builtins.print", side_effect=lambda s, **kw: output.append(str(s))), 

35 patch.dict("os.environ", env or {}, clear=True), 

36 pytest.raises(SystemExit), 

37 ): 

38 _run_hook() 

39 return output
