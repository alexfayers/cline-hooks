90def resolve_tooling_notes(workspace_roots: list[str]) -> list[str]: 

91 """Merge contributed tooling notes with the generic ecosystem note. 

92 

93 Args: 

94 workspace_roots: List of workspace root paths to search. 

95 

96 Returns: 

97 Ordered notes: the generic note (unless a contributor replaces it), then 

98 replacing contributor notes, then additive contributor notes. 

99 """ 

100 contributions = collect_contributions( 

101 EcosystemSpec.ecosystem_tooling_note, ToolingNote, workspace_roots=workspace_roots 

102 ) 

103 replacing = [c.note for c in contributions if c.replaces_generic] 

104 additive = [c.note for c in contributions if not c.replaces_generic] 

105 generic = None if replacing else _generic_tooling_note(workspace_roots) 

106 return [*([generic] if generic else []), *replacing, *additive]
